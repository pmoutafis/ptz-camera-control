async function checkBridge() { try { const res = await fetch('http://localhost:5001/health'); const data = await res.json(); log(`Local Bridge Status: CONNECTED (${data.os})`); } catch (e) { log(`Local Bridge Status: NOT DETECTED. Please launch PTZ-Bridge.`); } } async function moveMotor(axis, direction) { let p = axis === 'pan' ? direction * 10 : 0; let t = axis === 'tilt' ? direction * 10 : 0; let z = axis === 'zoom' ? direction * 10 : 0; if (axis === 'reset') { currentCoords = { pan: 0, tilt: 0, zoom: 0 }; } else { currentCoords.pan += p; currentCoords.tilt += t; currentCoords.zoom += z; } try { await fetch(`http://localhost:5001/ptz?p=${p}&t=${t}&z=${z}`, { method: 'POST' }); log(`Command sent: Pan ${currentCoords.pan}, Tilt ${currentCoords.tilt}, Zoom ${currentCoords.zoom}`); } catch (err) { log("Bridge error. Ensure PTZ-Bridge is running."); } }
